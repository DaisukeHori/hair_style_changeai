```typescript
// supabase/functions/pinterest-search/index.ts

import { serve } from 'https://deno.land/std@0.177.0/http/server.ts'
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const PINTEREST_API_URL = 'https://api.pinterest.com/v5'
const ACCESS_TOKEN = Deno.env.get('PINTEREST_ACCESS_TOKEN')

interface SearchRequest {
  query: string
  page_size?: number
  bookmark?: string
}

serve(async (req) => {
  try {
    const { query, page_size = 25, bookmark } = await req.json() as SearchRequest

    // Pinterest API呼び出し
    const params = new URLSearchParams({
      query,
      page_size: String(page_size),
    })
    if (bookmark) params.append('bookmark', bookmark)

    const response = await fetch(
      `${PINTEREST_API_URL}/pins/search?${params}`,
      {
        headers: {
          Authorization: `Bearer ${ACCESS_TOKEN}`,
        },
      }
    )

    if (!response.ok) {
      throw new Error(`Pinterest API error: ${response.status}`)
    }

    const data = await response.json()

    // レスポンス整形
    const items = data.items.map((pin: any) => ({
      pin_id: pin.id,
      title: pin.title,
      description: pin.description,
      image_url: pin.media?.images?.original?.url,
      source_url: pin.link,
    }))

    return new Response(
      JSON.stringify({
        items,
        bookmark: data.bookmark,
        has_more: !!data.bookmark,
      }),
      { headers: { 'Content-Type': 'application/json' } }
    )
  } catch (error) {
    return new Response(
      JSON.stringify({ error: error.message }),
      { status: 500, headers: { 'Content-Type': 'application/json' } }
    )
  }
})
```

### 7.2.3 画像取り込み処理

```typescript
// supabase/functions/pinterest-import/index.ts

serve(async (req) => {
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )

  const { pin_id, metadata } = await req.json()

  // 1. Pinterest から画像URL取得
  const pinData = await fetchPinDetails(pin_id)

  // 2. 画像をダウンロード
  const imageResponse = await fetch(pinData.image_url)
  const imageBlob = await imageResponse.blob()

  // 3. Supabase Storageにアップロード
  const fileName = `shared/${crypto.randomUUID()}.jpg`
  const { data: uploadData, error: uploadError } = await supabase.storage
    .from('hair-styles')
    .upload(fileName, imageBlob, {
      contentType: 'image/jpeg',
    })

  if (uploadError) throw uploadError

  // 4. サムネイル生成（別途処理）
  const thumbnailUrl = await generateThumbnail(fileName)

  // 5. DBに登録
  const { data: styleData, error: dbError } = await supabase
    .from('hair_styles')
    .insert({
      source: 'pinterest',
      source_url: pinData.source_url,
      image_url: uploadData.path,
      thumbnail_url: thumbnailUrl,
      ...metadata,
    })
    .select()
    .single()

  return new Response(JSON.stringify(styleData))
})
```

```typescript
// supabase/functions/simulation-create/index.ts

const NANOBANANA_API_URL = 'https://api.nanobanana.pro/v1'
const API_KEY = Deno.env.get('NANOBANANA_API_KEY')

interface SimulationRequest {
  customer_id: string
  customer_photo_url: string
  hair_style_id: string
  options?: {
    preserve_hair_color?: boolean
  }
}

serve(async (req) => {
  const supabase = createClient(/* ... */)
  const request: SimulationRequest = await req.json()

  // 1. スタイル画像URLを取得
  const { data: style } = await supabase
    .from('hair_styles')
    .select('image_url')
    .eq('id', request.hair_style_id)
    .single()

  const styleImageUrl = await getSignedUrl(style.image_url)

  // 2. NANOBANANA Pro API呼び出し
  const response = await fetch(`${NANOBANANA_API_URL}/simulate`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${API_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      source_image: { url: request.customer_photo_url },
      style_image: { url: styleImageUrl },
      options: {
        preserve_color: request.options?.preserve_hair_color ?? false,
        blend_mode: 'natural',
        output_format: 'png',
        output_size: { width: 1024, height: 1024 },
      },
    }),
  })

  const result = await response.json()

  if (result.status === 'processing') {
    // 非同期処理: ポーリングまたはWebhook待機
    return new Response(JSON.stringify({
      simulation_id: result.job_id,
      status: 'processing',
    }))
  }

  // 3. 結果画像を保存
  const resultImageUrl = await saveSimulationResult(
    result.result_image,
    request.customer_id
  )

  // 4. DBに記録
  const { data: simulation } = await supabase
    .from('style_simulations')
    .insert({
      customer_id: request.customer_id,
      hair_style_id: request.hair_style_id,
      original_photo_url: request.customer_photo_url,
      simulated_photo_url: resultImageUrl,
    })
    .select()
    .single()

  // 5. スタイルのシミュレーション数をインクリメント
  await supabase.rpc('increment_simulation_count', {
    style_id: request.hair_style_id,
  })

  return new Response(JSON.stringify({
    simulation_id: simulation.id,
    status: 'completed',
    result: {
      original_photo_url: request.customer_photo_url,
      simulated_photo_url: resultImageUrl,
    },
  }))
})
```

## 7.5 連携エラー対策

### 7.5.1 リトライ戦略

```typescript
// 指数バックオフによるリトライ
const fetchWithRetry = async (
  url: string,
  options: RequestInit,
  maxRetries = 3
) => {
  let lastError: Error

  for (let i = 0; i < maxRetries; i++) {
    try {
      const response = await fetch(url, options)

      if (response.status === 429) {
        // レート制限: 待機時間を計算
        const retryAfter = response.headers.get('Retry-After')
        const waitTime = retryAfter
          ? parseInt(retryAfter) * 1000
          : Math.pow(2, i) * 1000

        await sleep(waitTime)
        continue
      }

      if (!response.ok && response.status >= 500) {
        // サーバーエラー: リトライ
        throw new Error(`Server error: ${response.status}`)
      }

      return response
    } catch (error) {
      lastError = error
      await sleep(Math.pow(2, i) * 1000)
    }
  }

  throw lastError!
}
```

### 7.5.2 サーキットブレーカー

```typescript
// サーキットブレーカーパターン
class CircuitBreaker {
  private failures = 0
  private lastFailure = 0
  private state: 'CLOSED' | 'OPEN' | 'HALF_OPEN' = 'CLOSED'

  constructor(
    private threshold = 5,
    private timeout = 60000  // 1分
  ) {}

  async call<T>(fn: () => Promise<T>): Promise<T> {
    if (this.state === 'OPEN') {
      if (Date.now() - this.lastFailure > this.timeout) {
        this.state = 'HALF_OPEN'
      } else {
        throw new Error('Circuit is OPEN')
      }
    }

    try {
      const result = await fn()
      this.reset()
      return result
    } catch (error) {
      this.recordFailure()
      throw error
    }
  }

  private recordFailure() {
    this.failures++
    this.lastFailure = Date.now()
    if (this.failures >= this.threshold) {
      this.state = 'OPEN'
    }
  }

  private reset() {
    this.failures = 0
    this.state = 'CLOSED'
  }
}

// 使用例
const pinterestBreaker = new CircuitBreaker()
const result = await pinterestBreaker.call(() => fetchPinterest(query))
```
