# 第2章 状態管理設計

## 2.1 状態管理方針

### 2.1.1 状態の分類

| 分類 | 管理方法 | 用途 |
|------|---------|------|
| サーバー状態 | TanStack Query | API データ |
| グローバルUI状態 | Zustand | 認証、モーダル |
| ローカルUI状態 | useState | フォーム、一時的な状態 |
| URL状態 | React Router | ページ遷移、フィルタ |

## 2.2 Zustandストア設計

### 2.2.1 authStore

```typescript
// src/stores/authStore.ts

import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { User, Session } from '@supabase/supabase-js';
import type { Stylist } from '@/types/stylist';

interface AuthState {
  user: User | null;
  session: Session | null;
  stylist: Stylist | null;
  isLoading: boolean;
  isAuthenticated: boolean;
}

interface AuthActions {
  setUser: (user: User | null) => void;
  setSession: (session: Session | null) => void;
  setStylist: (stylist: Stylist | null) => void;
  setLoading: (loading: boolean) => void;
  signOut: () => void;
  initialize: () => Promise<void>;
}

export const useAuthStore = create<AuthState & AuthActions>()(
  persist(
    (set, get) => ({
      // State
      user: null,
      session: null,
      stylist: null,
      isLoading: true,
      isAuthenticated: false,

      // Actions
      setUser: (user) => set({
        user,
        isAuthenticated: !!user
      }),

      setSession: (session) => set({ session }),

      setStylist: (stylist) => set({ stylist }),

      setLoading: (isLoading) => set({ isLoading }),

      signOut: () => set({
        user: null,
        session: null,
        stylist: null,
        isAuthenticated: false,
      }),

      initialize: async () => {
        set({ isLoading: true });
        try {
          const { data: { session } } = await supabase.auth.getSession();
          if (session?.user) {
            const { data: stylist } = await supabase
              .from('stylists')
              .select('*')
              .eq('auth_user_id', session.user.id)
              .single();

            set({
              user: session.user,
              session,
              stylist,
              isAuthenticated: true,
            });
          }
        } finally {
          set({ isLoading: false });
        }
      },
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({
        // sessionのみ永続化
        session: state.session,
      }),
    }
  )
);

// セレクター
export const useCurrentUser = () => useAuthStore((state) => state.user);
export const useStylist = () => useAuthStore((state) => state.stylist);
export const useIsAuthenticated = () => useAuthStore((state) => state.isAuthenticated);
export const useSalonId = () => useAuthStore((state) => state.stylist?.salon_id);
```

### 2.2.2 uiStore

```typescript
// src/stores/uiStore.ts

import { create } from 'zustand';

interface Toast {
  id: string;
  type: 'success' | 'error' | 'warning' | 'info';
  title: string;
  message?: string;
}

interface ModalState {
  isOpen: boolean;
  type: string | null;
  props: Record<string, unknown>;
}

interface UIState {
  // サイドバー
  isSidebarOpen: boolean;
  toggleSidebar: () => void;
  setSidebarOpen: (open: boolean) => void;

  // モーダル
  modal: ModalState;
  openModal: (type: string, props?: Record<string, unknown>) => void;
  closeModal: () => void;

  // トースト
  toasts: Toast[];
  showToast: (toast: Omit<Toast, 'id'>) => void;
  removeToast: (id: string) => void;

  // ローディング
  globalLoading: boolean;
  setGlobalLoading: (loading: boolean) => void;
}

export const useUIStore = create<UIState>((set, get) => ({
  // サイドバー
  isSidebarOpen: false,
  toggleSidebar: () => set((state) => ({ isSidebarOpen: !state.isSidebarOpen })),
  setSidebarOpen: (isSidebarOpen) => set({ isSidebarOpen }),

  // モーダル
  modal: { isOpen: false, type: null, props: {} },
  openModal: (type, props = {}) => set({
    modal: { isOpen: true, type, props },
  }),
  closeModal: () => set({
    modal: { isOpen: false, type: null, props: {} },
  }),

  // トースト
  toasts: [],
  showToast: (toast) => {
    const id = Math.random().toString(36).substr(2, 9);
    set((state) => ({
      toasts: [...state.toasts, { ...toast, id }],
    }));
    // 自動削除
    setTimeout(() => {
      get().removeToast(id);
    }, 5000);
  },
  removeToast: (id) => set((state) => ({
    toasts: state.toasts.filter((t) => t.id !== id),
  })),

  // ローディング
  globalLoading: false,
  setGlobalLoading: (globalLoading) => set({ globalLoading }),
}));

// 便利なフック
export const useToast = () => {
  const showToast = useUIStore((state) => state.showToast);
  return {
    success: (title: string, message?: string) =>
      showToast({ type: 'success', title, message }),
    error: (title: string, message?: string) =>
      showToast({ type: 'error', title, message }),
    warning: (title: string, message?: string) =>
      showToast({ type: 'warning', title, message }),
    info: (title: string, message?: string) =>
      showToast({ type: 'info', title, message }),
  };
};
```

```typescript
// src/stores/customerStore.ts

import { create } from 'zustand';
import type { Customer } from '@/types/customer';

interface CustomerState {
  selectedCustomer: Customer | null;
  recentCustomers: Customer[];
  searchQuery: string;
}

interface CustomerActions {
  setSelectedCustomer: (customer: Customer | null) => void;
  addRecentCustomer: (customer: Customer) => void;
  setSearchQuery: (query: string) => void;
  clearSearch: () => void;
}

export const useCustomerStore = create<CustomerState & CustomerActions>((set) => ({
  // State
  selectedCustomer: null,
  recentCustomers: [],
  searchQuery: '',

  // Actions
  setSelectedCustomer: (selectedCustomer) => set({ selectedCustomer }),

  addRecentCustomer: (customer) => set((state) => {
    const filtered = state.recentCustomers.filter((c) => c.id !== customer.id);
    return {
      recentCustomers: [customer, ...filtered].slice(0, 10), // 最大10件
    };
  }),

  setSearchQuery: (searchQuery) => set({ searchQuery }),

  clearSearch: () => set({ searchQuery: '' }),
}));
```

### 2.3.2 カスタムフック

```typescript
// src/hooks/useCustomers.ts

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { queryKeys } from '@/services/queryKeys';
import * as customerApi from '@/services/api/customers';
import { useToast } from '@/stores/uiStore';
import type { Customer, CustomerFilters, CreateCustomerInput } from '@/types/customer';

export const useCustomers = (filters: CustomerFilters) => {
  return useQuery({
    queryKey: queryKeys.customers.list(filters),
    queryFn: () => customerApi.getCustomers(filters),
    staleTime: 1000 * 60 * 5, // 5分
  });
};

export const useCustomer = (id: string) => {
  return useQuery({
    queryKey: queryKeys.customers.detail(id),
    queryFn: () => customerApi.getCustomer(id),
    enabled: !!id,
  });
};

export const useCreateCustomer = () => {
  const queryClient = useQueryClient();
  const toast = useToast();

  return useMutation({
    mutationFn: (input: CreateCustomerInput) => customerApi.createCustomer(input),
    onSuccess: (newCustomer) => {
      // キャッシュを無効化
      queryClient.invalidateQueries({ queryKey: queryKeys.customers.lists() });
      toast.success('顧客を登録しました');
    },
    onError: (error) => {
      toast.error('登録に失敗しました', error.message);
    },
  });
};

export const useUpdateCustomer = () => {
  const queryClient = useQueryClient();
  const toast = useToast();

  return useMutation({
    mutationFn: ({ id, ...input }: { id: string } & Partial<Customer>) =>
      customerApi.updateCustomer(id, input),
    onSuccess: (updated, { id }) => {
      // 詳細キャッシュを更新
      queryClient.setQueryData(queryKeys.customers.detail(id), updated);
      // リストキャッシュを無効化
      queryClient.invalidateQueries({ queryKey: queryKeys.customers.lists() });
      toast.success('保存しました');
    },
    onError: (error) => {
      toast.error('保存に失敗しました', error.message);
    },
  });
};

export const useDeleteCustomer = () => {
  const queryClient = useQueryClient();
  const toast = useToast();

  return useMutation({
    mutationFn: (id: string) => customerApi.deleteCustomer(id),
    onSuccess: (_, id) => {
      queryClient.removeQueries({ queryKey: queryKeys.customers.detail(id) });
      queryClient.invalidateQueries({ queryKey: queryKeys.customers.lists() });
      toast.success('削除しました');
    },
    onError: (error) => {
      toast.error('削除に失敗しました', error.message);
    },
  });
};
```

### 2.3.3 シミュレーションフック

```typescript
// src/hooks/useSimulation.ts

import { useState } from 'react';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { queryKeys } from '@/services/queryKeys';
import * as simulationApi from '@/services/api/simulation';
import { useToast } from '@/stores/uiStore';
import type { SimulationRequest, SimulationResult } from '@/types/simulation';

export const useSimulation = () => {
  const queryClient = useQueryClient();
  const toast = useToast();
  const [progress, setProgress] = useState(0);

  const mutation = useMutation({
    mutationFn: async (request: SimulationRequest) => {
      setProgress(10);

      // シミュレーション開始
      const { simulation_id, status } = await simulationApi.createSimulation(request);
      setProgress(30);

      // ポーリングで結果を待つ
      if (status === 'processing') {
        let result: SimulationResult | null = null;
        let attempts = 0;
        const maxAttempts = 30; // 最大30秒

        while (!result && attempts < maxAttempts) {
          await new Promise((resolve) => setTimeout(resolve, 1000));
          const status = await simulationApi.getSimulationStatus(simulation_id);
          setProgress(30 + (attempts / maxAttempts) * 60);

          if (status.status === 'completed') {
            result = status.result;
          } else if (status.status === 'failed') {
            throw new Error(status.error || 'シミュレーションに失敗しました');
          }
          attempts++;
        }

        if (!result) {
          throw new Error('タイムアウトしました');
        }

        setProgress(100);
        return result;
      }

      setProgress(100);
      return null;
    },
    onSuccess: (result, { customer_id }) => {
      if (result) {
        // シミュレーション履歴を更新
        queryClient.invalidateQueries({
          queryKey: queryKeys.simulations.byCustomer(customer_id),
        });
        toast.success('シミュレーション完了');
      }
    },
    onError: (error: Error) => {
      setProgress(0);
      toast.error('シミュレーション失敗', error.message);
    },
    onSettled: () => {
      // 少し待ってからプログレスをリセット
      setTimeout(() => setProgress(0), 1000);
    },
  });

  return {
    simulate: mutation.mutate,
    isLoading: mutation.isPending,
    progress,
    result: mutation.data,
    error: mutation.error,
  };
};
```

```typescript
// src/hooks/useCustomerForm.ts

import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';

export const customerSchema = z.object({
  first_name: z.string().min(1, '名前を入力してください'),
  last_name: z.string().min(1, '姓を入力してください'),
  first_name_kana: z.string().optional(),
  last_name_kana: z.string().optional(),
  gender: z.enum(['male', 'female', 'other']).optional(),
  birth_date: z.string().optional(),
  phone: z.string().regex(/^0\d{9,10}$/, '正しい電話番号を入力してください').optional().or(z.literal('')),
  email: z.string().email('正しいメールアドレスを入力してください').optional().or(z.literal('')),
  notes: z.string().optional(),
}).refine((data) => data.phone || data.email, {
  message: '電話番号またはメールアドレスを入力してください',
  path: ['phone'],
});

export type CustomerFormData = z.infer<typeof customerSchema>;

export const useCustomerForm = (defaultValues?: Partial<CustomerFormData>) => {
  return useForm<CustomerFormData>({
    resolver: zodResolver(customerSchema),
    defaultValues: {
      first_name: '',
      last_name: '',
      first_name_kana: '',
      last_name_kana: '',
      gender: undefined,
      birth_date: '',
      phone: '',
      email: '',
      notes: '',
      ...defaultValues,
    },
  });
};
```
