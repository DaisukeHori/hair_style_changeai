## 4.1 マイグレーション

### 4.1.1 初期スキーマ

```sql
-- supabase/migrations/001_initial_schema.sql

-- 拡張機能
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 更新日時自動更新関数
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- salons テーブル
CREATE TABLE salons (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(100) NOT NULL,
    address VARCHAR(255),
    phone VARCHAR(20),
    email VARCHAR(255) UNIQUE,
    business_hours JSONB DEFAULT '{}',
    logo_url VARCHAR(500),
    settings JSONB DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TRIGGER update_salons_updated_at
    BEFORE UPDATE ON salons
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- stylists テーブル
CREATE TABLE stylists (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    salon_id UUID NOT NULL REFERENCES salons(id) ON DELETE CASCADE,
    auth_user_id UUID REFERENCES auth.users(id),
    name VARCHAR(50) NOT NULL,
    nickname VARCHAR(50),
    role VARCHAR(20) NOT NULL DEFAULT 'stylist'
        CHECK (role IN ('owner', 'manager', 'stylist', 'assistant')),
    profile_image_url VARCHAR(500),
    specialties JSONB DEFAULT '[]',
    bio TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_stylists_salon ON stylists(salon_id);
CREATE INDEX idx_stylists_auth ON stylists(auth_user_id);
CREATE TRIGGER update_stylists_updated_at
    BEFORE UPDATE ON stylists
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- customers テーブル
CREATE TABLE customers (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    salon_id UUID NOT NULL REFERENCES salons(id) ON DELETE CASCADE,
    customer_code VARCHAR(20) NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    first_name_kana VARCHAR(50),
    last_name_kana VARCHAR(50),
    gender VARCHAR(10) CHECK (gender IN ('male', 'female', 'other')),
    birth_date DATE,
    phone VARCHAR(20),
    email VARCHAR(255),
    line_id VARCHAR(100),
    address VARCHAR(255),
    occupation VARCHAR(50),
    referral_source VARCHAR(50),
    notes TEXT,
    profile_photo_url VARCHAR(500),
    last_visit_date DATE,
    visit_count INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(salon_id, customer_code)
);

CREATE INDEX idx_customers_salon ON customers(salon_id);
CREATE INDEX idx_customers_name ON customers(salon_id, last_name_kana);
CREATE INDEX idx_customers_phone ON customers(salon_id, phone);
CREATE TRIGGER update_customers_updated_at
    BEFORE UPDATE ON customers
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- 顧客番号自動生成
CREATE OR REPLACE FUNCTION generate_customer_code()
RETURNS TRIGGER AS $$
DECLARE next_code INTEGER;
BEGIN
    SELECT COALESCE(MAX(CAST(SUBSTRING(customer_code FROM 2) AS INTEGER)), 0) + 1
    INTO next_code FROM customers WHERE salon_id = NEW.salon_id;
    NEW.customer_code = 'C' || LPAD(next_code::TEXT, 5, '0');
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER set_customer_code
    BEFORE INSERT ON customers
    FOR EACH ROW WHEN (NEW.customer_code IS NULL)
    EXECUTE FUNCTION generate_customer_code();
```

### 4.1.2 髪質・施術テーブル

-- supabase/migrations/002_hair_visits.sql

-- customer_hair_profiles
CREATE TABLE customer_hair_profiles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    customer_id UUID NOT NULL REFERENCES customers(id) ON DELETE CASCADE UNIQUE,
    hair_type VARCHAR(20),
    hair_thickness VARCHAR(20),
    hair_volume VARCHAR(20),
    hair_damage_level INTEGER CHECK (hair_damage_level BETWEEN 1 AND 5),
    scalp_type VARCHAR(20),
    scalp_sensitivity INTEGER CHECK (scalp_sensitivity BETWEEN 1 AND 5),
    gray_hair_percentage INTEGER CHECK (gray_hair_percentage BETWEEN 0 AND 100),
    allergies JSONB DEFAULT '{"items": [], "notes": ""}',
    previous_treatments JSONB DEFAULT '[]',
    notes TEXT,
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    updated_by UUID REFERENCES stylists(id)
);

-- visits
CREATE TABLE visits (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    customer_id UUID NOT NULL REFERENCES customers(id) ON DELETE CASCADE,
    stylist_id UUID NOT NULL REFERENCES stylists(id),
    salon_id UUID NOT NULL REFERENCES salons(id),
    visit_date DATE NOT NULL,
    check_in_time TIME,
    check_out_time TIME,
    status VARCHAR(20) DEFAULT 'reserved',
    total_amount INTEGER,
    payment_method VARCHAR(20),
    notes TEXT,
    next_visit_suggestion TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_visits_customer ON visits(customer_id);
CREATE INDEX idx_visits_salon_date ON visits(salon_id, visit_date DESC);

-- visit_services
CREATE TABLE visit_services (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    visit_id UUID NOT NULL REFERENCES visits(id) ON DELETE CASCADE,
    service_type VARCHAR(20) NOT NULL,
    service_name VARCHAR(100) NOT NULL,
    price INTEGER,
    duration_minutes INTEGER,
    details JSONB DEFAULT '{}',
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- visit_photos
CREATE TABLE visit_photos (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    visit_id UUID NOT NULL REFERENCES visits(id) ON DELETE CASCADE,
    photo_type VARCHAR(20) NOT NULL,
    photo_url VARCHAR(500) NOT NULL,
    angle VARCHAR(20),
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

### 4.1.3 スタイル・シミュレーションテーブル

```sql
-- supabase/migrations/003_styles_simulations.sql

-- hair_styles
CREATE TABLE hair_styles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    salon_id UUID REFERENCES salons(id) ON DELETE CASCADE,
    source VARCHAR(20) NOT NULL,
    source_url VARCHAR(500),
    image_url VARCHAR(500) NOT NULL,
    thumbnail_url VARCHAR(500),
    title VARCHAR(200),
    description TEXT,
    tags JSONB DEFAULT '[]',
    hair_length VARCHAR(20),
    hair_color JSONB DEFAULT '{}',
    face_shapes JSONB DEFAULT '[]',
    gender_target VARCHAR(10),
    is_active BOOLEAN DEFAULT TRUE,
    view_count INTEGER DEFAULT 0,
    simulation_count INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_styles_tags ON hair_styles USING gin(tags);
CREATE INDEX idx_styles_active ON hair_styles(is_active) WHERE is_active = TRUE;

-- style_simulations
CREATE TABLE style_simulations (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    customer_id UUID NOT NULL REFERENCES customers(id) ON DELETE CASCADE,
    hair_style_id UUID NOT NULL REFERENCES hair_styles(id),
    original_photo_url VARCHAR(500) NOT NULL,
    simulated_photo_url VARCHAR(500) NOT NULL,
    is_liked BOOLEAN,
    stylist_notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    created_by UUID REFERENCES stylists(id)
);

CREATE INDEX idx_simulations_customer ON style_simulations(customer_id);

-- customer_favorite_styles
CREATE TABLE customer_favorite_styles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    customer_id UUID NOT NULL REFERENCES customers(id) ON DELETE CASCADE,
    hair_style_id UUID NOT NULL REFERENCES hair_styles(id),
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(customer_id, hair_style_id)
);

-- appointments
CREATE TABLE appointments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    customer_id UUID NOT NULL REFERENCES customers(id) ON DELETE CASCADE,
    stylist_id UUID NOT NULL REFERENCES stylists(id),
    salon_id UUID NOT NULL REFERENCES salons(id),
    appointment_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    services JSONB DEFAULT '[]',
    status VARCHAR(20) DEFAULT 'pending',
    reminder_sent BOOLEAN DEFAULT FALSE,
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_appointments_date ON appointments(salon_id, appointment_date);
```

### 4.1.4 RLSポリシー

```sql
-- supabase/migrations/004_rls_policies.sql

-- RLS有効化
ALTER TABLE salons ENABLE ROW LEVEL SECURITY;
ALTER TABLE stylists ENABLE ROW LEVEL SECURITY;
ALTER TABLE customers ENABLE ROW LEVEL SECURITY;
ALTER TABLE customer_hair_profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE visits ENABLE ROW LEVEL SECURITY;
ALTER TABLE visit_services ENABLE ROW LEVEL SECURITY;
ALTER TABLE visit_photos ENABLE ROW LEVEL SECURITY;
ALTER TABLE hair_styles ENABLE ROW LEVEL SECURITY;
ALTER TABLE style_simulations ENABLE ROW LEVEL SECURITY;
ALTER TABLE customer_favorite_styles ENABLE ROW LEVEL SECURITY;
ALTER TABLE appointments ENABLE ROW LEVEL SECURITY;

-- ヘルパー関数
CREATE OR REPLACE FUNCTION get_user_salon_id()
RETURNS UUID AS $$
BEGIN
    RETURN (SELECT salon_id FROM stylists WHERE auth_user_id = auth.uid());
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- customersポリシー
CREATE POLICY "salon_customers_select" ON customers
    FOR SELECT USING (salon_id = get_user_salon_id());

CREATE POLICY "salon_customers_insert" ON customers
    FOR INSERT WITH CHECK (salon_id = get_user_salon_id());

CREATE POLICY "salon_customers_update" ON customers
    FOR UPDATE USING (salon_id = get_user_salon_id());

CREATE POLICY "salon_customers_delete" ON customers
    FOR DELETE USING (salon_id = get_user_salon_id());

-- hair_stylesポリシー（共有スタイルも閲覧可能）
CREATE POLICY "styles_select" ON hair_styles
    FOR SELECT USING (salon_id IS NULL OR salon_id = get_user_salon_id());

-- 他テーブルも同様に設定
```

## 4.2 RPC関数

```sql
-- supabase/migrations/005_rpc_functions.sql

-- シミュレーション数インクリメント
CREATE OR REPLACE FUNCTION increment_simulation_count(style_id UUID)
RETURNS void AS $$
BEGIN
    UPDATE hair_styles
    SET simulation_count = simulation_count + 1
    WHERE id = style_id;
END;
$$ LANGUAGE plpgsql;

-- 閲覧数インクリメント
CREATE OR REPLACE FUNCTION increment_view_count(style_id UUID)
RETURNS void AS $$
BEGIN
    UPDATE hair_styles
    SET view_count = view_count + 1
    WHERE id = style_id;
END;
$$ LANGUAGE plpgsql;

-- 顧客統計更新
CREATE OR REPLACE FUNCTION update_customer_visit_stats()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'INSERT' OR (TG_OP = 'UPDATE' AND OLD.status != 'completed' AND NEW.status = 'completed') THEN
        UPDATE customers SET
            visit_count = visit_count + 1,
            last_visit_date = NEW.visit_date
        WHERE id = NEW.customer_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_customer_stats
    AFTER INSERT OR UPDATE ON visits
    FOR EACH ROW EXECUTE FUNCTION update_customer_visit_stats();
```
